pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                echo 'Building the project...'
                bat 'python -m py_compile app.py'
            }
        }
        stage('Test') {
            steps {
                echo 'Running automated unit tests...'
                bat 'python -m unittest test_app.py'
            }
        }
    }
    post {
        success {
            echo 'CI Pipeline executed successfully!'
        }
        failure {
            echo 'CI Pipeline failed.'
        }
    }
}
